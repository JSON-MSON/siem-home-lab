import json
from collections import Counter
def rd(p):
    with open(p, errors="replace") as f:
        for l in f:
            try: yield json.loads(l)
            except Exception: pass
A = "/var/ossec/logs/alerts/2026/"
first = None; c = Counter()
for a in rd(A + "Jul/ossec-alerts-24.json"):
    ts = a["timestamp"][:19]; rid = a["rule"]["id"]
    if rid == "100010" and first is None: first = a["timestamp"]
    if "2026-07-24T19:25:00" <= ts < "2026-07-24T20:09:00" and a.get("data", {}).get("srcip") == "198.51.100.128":
        c[(rid, a["rule"].get("description"))] += 1
print("## Jul 24 first rule 100010 alert of the day:", first)
print("## Jul 24 rules fired by 198.51.100.128 between 19:25 and 20:09 (before the fixed rule loaded)")
for k, v in sorted(c.items()): print(k, v)
c2 = Counter()
for a in rd(A + "Jul/ossec-alerts-28.json"):
    ts = a["timestamp"][:19]
    if "2026-07-28T13:28:00" <= ts <= "2026-07-28T13:48:00": c2[(a["rule"]["id"], a["rule"].get("description"))] += 1
print("## Jul 28 rules fired 13:28-13:48")
